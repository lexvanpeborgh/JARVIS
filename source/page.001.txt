const emptyState = () => ({
  version: 3,
  userName: '',
  sessionId: crypto.randomUUID(),
  messages: [],
  memories: [],
  tasks: [],
  events: [],
  notes: '',
  voiceArchive: [],
});

function uid() { return crypto.randomUUID(); }
function safeState(raw) {
  const base = emptyState();
  if (!raw || typeof raw !== 'object') return base;
  return {
    ...base,
    ...raw,
    version:3,
    messages: Array.isArray(raw.messages) ? raw.messages : [],
    memories: Array.isArray(raw.memories) ? raw.memories : [],
    tasks: Array.isArray(raw.tasks) ? raw.tasks : [],
    events: Array.isArray(raw.events) ? raw.events : [],
    voiceArchive: Array.isArray(raw.voiceArchive) ? raw.voiceArchive : [],
  };
}
function words(text) {
  return new Set(String(text || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').match(/[a-z0-9à-ÿ]{3,}/gi) || []);
}
function relevantMemories(memories, query, limit = 10) {
  const q = words(query);
  return memories.map(m => {
    const w = words(m.text); let score = 0;
    q.forEach(x => { if (w.has(x)) score += 3; });
    if (['preference','goal','profile','routine','project'].includes(m.category)) score += 1;
    return { m, score };
  }).sort((a,b) => b.score - a.score || String(b.m.createdAt).localeCompare(String(a.m.createdAt))).slice(0, limit).map(x => x.m);
}
function historyMatches(messages, query, currentSession, limit = 8) {
  const q = words(query);
  return messages.filter(m => m.sessionId !== currentSession).map(m => {
    const w = words(m.text); let score = 0;
    q.forEach(x => { if (w.has(x)) score += 1; });
    return { m, score };
  }).filter(x => x.score > 0).sort((a,b) => b.score - a.score).slice(0, limit).map(x => x.m);
}
function formatDate(value) {
  try { return new Intl.DateTimeFormat('nl-BE', { weekday:'short', day:'numeric', month:'short', hour:'2-digit', minute:'2-digit' }).format(new Date(value)); }
  catch { return String(value); }
}
function readFile(file) {
  return new Promise((resolve, reject) => {
    const r = new FileReader();
    r.onload = () => { const result = String(r.result || ''); resolve(result.includes(',') ? result.split(',')[1] : result); };
    r.onerror = () => reject(r.error);
    r.readAsDataURL(file);
  });
}
function bufferToBase64(buffer) {
  const bytes = new Uint8Array(buffer); let binary = ''; const size = 0x8000;
  for (let i=0;i<bytes.length;i+=size) binary += String.fromCharCode(...bytes.subarray(i, i+size));
  return btoa(binary);
}
function base64ToBytes(value) {
  const bin = atob(value); const out = new Uint8Array(bin.length);
  for (let i=0;i<bin.length;i++) out[i] = bin.charCodeAt(i);
  return out;
}
